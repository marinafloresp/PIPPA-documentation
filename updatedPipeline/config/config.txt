## ===========================================================================##
##                      ATAC-seq pipeline: Example config file                ##
## ===========================================================================##
##                                                                            ||
## This is an example config file for the ATAC-sequencing pipeline.           ||
## This file and the R config file are expected to be located in the project's||
## directory.                                                                 ||
## The full path to this file should be specified as the first command line   ||
## argument when running the pipeline's scripts.                              ||
## Please check all the paths as they should be customised by the user.       ||
##                                                                            ||
## ===========================================================================##

## ========================= ##
|| Project set up filepaths  ||
## ========================= ##

## Specify the name of the project's directory ##
export PROJECT="project name"
## Specify the full project's directory ##
export MAIN_DIR="/path/to/project/folder/${PROJECT}"

export CONFIGR="${MAIN_DIR}/config.r"
export CONDAFILE="${MAIN_DIR}/environment.txt"

## =========================================== ##
|| Scripts and software filepaths and versions ||
## =========================================== ##

## Specify full path to pipeline's scripts directory ##
export SCRIPTS_DIR="/path/to/these/scripts/folder"

export SUB_SCRIPTS_DIR="${SCRIPTS_DIR}/subscripts"
export RSCRIPTS_DIR="${SCRIPTS_DIR}/Rscripts"
export LOG_DIR="${SCRIPTS_DIR}/logFiles"

## Specify full path to conda environment ##
export CONDA="/path/tp/.conda/envs/folder"
export CONDA_ENV="/path/to/conda/etc/folder"

## =================== ##
|| Main data filepaths ||
## =================== ##

## Project's main data and results directories ##
export META_DIR="${MAIN_DIR}/0_metadata"
export RAWDATADIR="${MAIN_DIR}/1_raw/"
export FASTQCDIR="${MAIN_DIR}/1_raw/fastqc"
export TRIM_DIR="${MAIN_DIR}/2_trimmed"
export ALIGNED_DIR="${MAIN_DIR}/3_aligned"
export PEAK_DIR="${MAIN_DIR}/4_calledPeaks"
export PEAKCOUNTS="${MAIN_DIR}/5_countPeaks"
export ANALYSIS_DIR="${MAIN_DIR}/6_analysis"

## Specify full path to SNP imputation data ##
export GENODIR="/path/to/SNParray/ImputationData"

## ===================== ##
|| Output data filepaths ||
## ===================== ##

## Peak calling output directories ##
export PEAK_DIR_SAMPLES="${PEAK_DIR}/path/to/samples/peaks/output/samplePeaks/"
export PEAK_DIR_GROUPS="${PEAK_DIR}/path/to/groups/peaks/output/groupCheck/"
export PEAK_DIR_SEXCHR="${PEAK_DIR}/path/to/peaks/sex/chromosomes/output/sexChr/"
export PEAK_DIR_C="${PEAK_DIR}/path/to/peaksCuration/peaksAnalysis/"
export CURATED_PEAKS="${PEAK_DIR}/path/to/peaksCuration/curatedPeaks/"
export PEAK_DIR_CUTOFF="${PEAK_DIR}/path/to/peaksCuration/cutoff/"

## Counts in peaks output directories ##

export COUNTS_DIR_SEXCHR="${PEAKCOUNTS}/sexChr/"
export COUNTS_DIR_GROUPS="${PEAKCOUNTS}/checkCTCounts/"
export COUNTS_DIR_DOWN="${PEAKCOUNTS}/caseControl/"

## ========================= ##
|| Reference data filepaths  ||
## ========================= ##

## Specify full path to references folder ##
export REFERENCES_DIR="/path/to/references/folder"

## Specify the path to each of the reference datasets, files or folders ##
export REFGENOME="${REFERENCES_DIR}/reference_genome"
export BLACKLIST="${REFERENCES_DIR}/blacklisted_peak_regions.bed"
export XCHRBED="${REFERENCES_DIR}/X_chr_genes"
export PAR="${REFERENCES_DIR}/pseudoautosomal_regions"
export GENOMEFASTA="${REFERENCES_DIR}/Whole_genome_fasta/"
export KGREF="${REFERENCES_DIR}/reference_1000G/"
export SNP="${REFERENCES_DIR}/SNP_file"
export UCSCRef="${REFERENCES_DIR}/UCSC/"
export GENOMESIZE="${REFERENCES_DIR}/genome.info"
export SEGREFGENOME="${REFERENCES_DIR}/genome.char.fa"
export REFTSS="${REFERENCES_DIR}/hg38_refseq_TSS.bed"

## ======================= ##
||    TRIMMING SETTINGS    ||
## ======================= ##

## Adapter sequences to trim ##
## Fasta file with adapters to be trimmed at STEP 1.2 ##
export ADAPTERS_FILE="/path/to/fasta/adapters.fa"

## Trimming optional parameters (to be specified by user as desired) ## 
export MAX_LEN_R1=50
export MAX_LEN_R2=50
export MIN_LEN=27
export ADAP_R1=CTGTCTCTTATACACATCTCCGAGCCCACGAGAC
export ADAP_R2=CTGTCTCTTATACACATCTGACGCTGCCGACGA

## ======================= ##
||   ALIGNMENT SETTINGS    ||
## ======================= ##

## parameters (to be specified by user as desired) ## 
export multimap=4
export p_threads=10
export max_frag_len=2000
export mapq_thres=30

## cell types (ordered alphabetically)
export CELLTYPES=("IRF8" "NEUN" "SOX" "TN" )
export CLINICAL_GROUPS=("case" "control")

## =========================== ##
||    PEAK CALLING SETTINGS    ||
## =========================== ##

## to be changed by user ##
export macs_sample_thres=5e-2
export macs_group_thres=5e-2
export macs_sexchr_thres=1e-4
